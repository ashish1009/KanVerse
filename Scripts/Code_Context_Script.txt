python3 -c '
import os

target_dirs = ["KanVerse_Engine", "KanVerse_App"]
skip_folder = "Vendors"
valid_exts = {".h", ".hpp", ".cpp", ".mm", ".metal", ".pch"}
output_file = "Documents/KanVerse_Full_Codebase.txt"

def process_code(text):
    # Optional: If you also want to keep empty lines so comment spacing/formatting 
    # remains untouched, return text directly. Otherwise, you can keep the 
    # line filtering below.
    lines = text.splitlines()
    non_empty_lines = [line for line in lines if line.strip() != ""]
    return "\n".join(non_empty_lines)

with open(output_file, "w", encoding="utf-8") as outfile:
    for target in target_dirs:
        if not os.path.exists(target):
            continue
        for root, dirs, files in sorted(os.walk(target)):
            path_parts = os.path.normpath(root).split(os.sep)
            if skip_folder in path_parts:
                continue
            
            dirs[:] = [d for d in dirs if d != skip_folder]
            
            for file in sorted(files):
                ext = os.path.splitext(file)[1]
                if ext in valid_exts:
                    full_path = os.path.join(root, file)
                    # Write file header
                    outfile.write(f"// FILE: {full_path}\n")
                    try:
                        with open(full_path, "r", encoding="utf-8", errors="replace") as infile:
                            content = infile.read()
                            final_content = process_code(content)
                            outfile.write(final_content + "\n")
                    except Exception as e:
                        outfile.write(f"// Error reading file: {e}\n")

print(f"Extraction complete! Generated {output_file}")
